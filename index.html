function executeCommand(command) { if (!isPaused) { document.getElementById('status').innerText = "🟢 處理中..."; document.getElementById('status').style.color = "#0056b3"; } var url = document.getElementById('webAppUrl').value.trim(); if (!url.includes("script.google.com") || url.includes(".......")) { log("⚠️ 錯誤：請先在上方框框貼上正確的 Web App 網址！"); return; } localStorage.setItem("myWebAppUrl", url); var fetchUrl = url + "?command=" + encodeURIComponent(command) + "¤tAssignment=" + encodeURIComponent(activeAssignment); // 💡 即時印出連線提示，讓您知道系統正在運作 log("⏳ 正在傳送指令至 Google 雲端..."); fetch(fetchUrl) .then(res => { if (!res.ok) throw new Error("HTTP 錯誤 " + res.status); return res.json(); }) .then(data => { log("💻 電腦回應：" + data.speak); if (data.speak !== "收到指令但未辨識成功") speakOut(data.speak); if (data.activeAssignment) { activeAssignment = data.activeAssignment; document.getElementById('currentTask').innerText = activeAssignment; } if (!isSpeaking && !isPaused) { document.getElementById('status').innerText = "🟢 聆聽中..."; document.getElementById('status').style.color = "#28a745"; } }) .catch(err => { log("❌ 連線失敗/後台回應異常：" + err.message); if (!isPaused) { document.getElementById('status').innerText = "🟢 聆聽中..."; document.getElementById('status').style.color = "#28a745"; } }); }
